## Basic Usage

```typescript
import { Plugin, Provider } from '@frontmcp/sdk';

// Provider for the plugin
@Provider({ name: 'CacheService' })
export class CacheService {
  private cache = new Map<string, string>();

  async get(key: string) {
    return this.cache.get(key);
  }

  async set(key: string, value: string) {
    this.cache.set(key, value);
  }
}

// Plugin definition
@Plugin({
  name: 'cache',
  description: 'In-memory caching for tools',
  providers: [CacheService],
  exports: [CacheService],
})
export class CachePlugin {}
```

A plugin's `providers` are private to the plugin: its own tools, hooks and nested plugins resolve them. List a provider in `exports` as well so the tools of the app that installs the plugin can call `this.get(CacheService)`. A decorated class is its own DI token.

## Signature

```typescript
function Plugin(providedMetadata: PluginMetadata): ClassDecorator
```

## Configuration Options

### Required Properties

| Property | Type     | Description              |
| -------- | -------- | ------------------------ |
| `name`   | `string` | Unique plugin identifier |

### Optional Properties

| Property      | Type                | Description        |
| ------------- | ------------------- | ------------------ |
| `id`          | `string`            | Stable identifier  |
| `description` | `string`            | Plugin description |
| `scope`       | `'app' \| 'server'` | Plugin scope level |

### Components

| Property    | Type             | Description                                            |
| ----------- | ---------------- | ------------------------------------------------------ |
| `providers` | `ProviderType[]` | Plugin-scoped providers                                |
| `exports`   | `ProviderType[]` | Providers re-exported to the host app or other plugins |
| `tools`     | `ToolType[]`     | Plugin tools                                           |
| `resources` | `ResourceType[]` | Plugin resources                                       |
| `prompts`   | `PromptType[]`   | Plugin prompts                                         |
| `skills`    | `SkillType[]`    | Plugin skills                                          |
| `adapters`  | `AdapterType[]`  | Framework adapters                                     |
| `plugins`   | `PluginType[]`   | Nested plugins                                         |

### Extensions

| Property            | Type                 | Description                                                                                                                                                                                                                    |
| ------------------- | -------------------- | ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------ |
| `contextExtensions` | `ContextExtension[]` | Add properties to context classes                                                                                                                                                                                              |
| `enforcesMetadata`  | `string[]`           | Entry metadata keys this plugin's hooks enforce. A server where an entry declares one and no such plugin reaches it refuses to start (`UnenforcedMetadataError`)                                                               |
| `dynamicSkills`     | `boolean`            | The plugin registers skills at runtime (`scope.skills.registerSkillContent`). The server serves the skills methods, the SEP-2640 capability and `skill://` resources from startup, answering with the skills registered so far |

## Context Extensions

Extend all context classes with custom properties:

```typescript
import { Plugin, Provider, type Token } from '@frontmcp/sdk';

// 1. Define the accessor
@Provider({ name: 'MemoryAccessor' })
export class MemoryAccessor {
  private readonly values = new Map<string, string>();

  async get(key: string): Promise<string | undefined> {
    return this.values.get(key);
  }

  async set(key: string, value: string): Promise<void> {
    this.values.set(key, value);
  }
}

// 2. Define a token for it: a Symbol typed with Token<T>
export const MemoryAccessorToken: Token<MemoryAccessor> = Symbol('MemoryAccessor');

const memoryAccessorProvider = {
  name: 'MemoryAccessorToken',
  provide: MemoryAccessorToken,
  inject: () => [MemoryAccessor] as const,
  useFactory: (accessor: MemoryAccessor) => accessor,
};

// 3. Augment the context types (here or in a separate file the project compiles)
declare module '@frontmcp/sdk' {
  interface ExecutionContextBase {
    readonly memory: MemoryAccessor;
  }
}

// 4. Define the plugin with the context extension
@Plugin({
  name: 'memory',
  description: 'In-memory key-value store shared by tools',
  providers: [MemoryAccessor, memoryAccessorProvider],
  exports: [memoryAccessorProvider],
  contextExtensions: [
    {
      property: 'memory',
      token: MemoryAccessorToken,
      errorMessage: 'Memory plugin not installed',
    },
  ],
})
export class MemoryPlugin {}
```

`this.memory` resolves `MemoryAccessorToken` from the calling tool's providers, so the plugin exports the provider that binds it. The token can also be the provider class itself (`token: MemoryAccessor`, with `exports: [MemoryAccessor]`).

Now in tools:

```typescript
import { Tool, ToolContext } from '@frontmcp/sdk';

@Tool({ name: 'my_tool', inputSchema: {} })
class MyTool extends ToolContext {
  async execute() {
    // Access extended property
    await this.memory.set('lastRun', new Date().toISOString());
    const lastRun = await this.memory.get('lastRun');
    return { lastRun };
  }
}
```

<Note>
The official [RememberPlugin](/frontmcp/plugins/remember-plugin) already declares `this.remember`. A plugin of your own needs a property name no other installed plugin uses.
</Note>

## Plugin Scopes

### App Scope (Default)

The plugin's hooks run only for the entries of the app that installs it:

```typescript
@Plugin({
  name: 'app-plugin',
  scope: 'app', // Default
})
class AppPlugin {}
```

### Server Scope

The plugin's hooks register at the gateway/server level and run for every app's entries, even when the plugin is installed on one `@App`. Its providers and context extensions stay with the app that installs it:

```typescript
@Plugin({
  name: 'server-plugin',
  scope: 'server',
})
class ServerPlugin {}
```

<Warning>
Server-scoped plugins cannot be used in standalone apps. They require a gateway configuration.
</Warning>

## Nested Plugins

Plugins can include other plugins:

```typescript
@Plugin({
  name: 'analytics',
  plugins: [MetricsPlugin, LoggingPlugin],
  providers: [AnalyticsService],
})
class AnalyticsPlugin {}
```

The providers a `DynamicPlugin` derives from its options (`dynamicProviders(options)` and `init({ providers })`) are registered before its nested plugins are built. A nested plugin can inject them, for example with `NestedPlugin.init({ inject: () => [ConfigToken], useFactory })`. This holds for both `init(options)` and `init({ inject, useFactory })`.

## Using Plugins

### In Apps

```typescript
@App({
  name: 'my-app',
  plugins: [CachePlugin, RememberPlugin],
  tools: [MyTool],
})
class MyApp {}
```

### In Server

```typescript
@FrontMcp({
  info: { name: 'Server', version: '1.0.0' },
  apps: [MyApp],
  plugins: [GlobalLoggingPlugin], // Server-level plugins
})
class Server {}
```

## Plugin with Tools

```typescript
import { Plugin, Tool, ToolContext } from '@frontmcp/sdk';
import { z } from '@frontmcp/sdk';

@Tool({
  name: 'send_elicitation_result',
  description: 'Send user response to pending elicitation',
  inputSchema: {
    elicitId: z.string(),
    response: z.record(z.string(), z.unknown()),
  },
})
class SendElicitationResultTool extends ToolContext {
  async execute(input: { elicitId: string; response: Record<string, unknown> }) {
    // Plugin-provided tool implementation
    const elicitation = this.get(ElicitationService);
    await elicitation.resolve(input.elicitId, input.response);
    return { delivered: true };
  }
}

@Plugin({
  name: 'elicitation',
  tools: [SendElicitationResultTool],
  providers: [ElicitationService],
})
export class ElicitationPlugin {}
```

## Full Example

```typescript
import { Plugin, Provider, Tool, ToolContext, App, FrontMcp } from '@frontmcp/sdk';
import { z } from '@frontmcp/sdk';

// Services
@Provider({ name: 'ApprovalService' })
class ApprovalService {
  private approvals = new Map<string, boolean>();

  approve(toolName: string) {
    this.approvals.set(toolName, true);
  }

  isApproved(toolName: string): boolean {
    return this.approvals.get(toolName) === true;
  }

  revoke(toolName: string) {
    this.approvals.delete(toolName);
  }
}

@Provider({ name: 'ApprovalAccessor' })
class ApprovalAccessor {
  constructor(private service: ApprovalService) {}

  async isApproved(toolName: string): Promise<boolean> {
    return this.service.isApproved(toolName);
  }

  async approve(toolName: string): Promise<void> {
    this.service.approve(toolName);
  }
}

// Module augmentation
declare module '@frontmcp/sdk' {
  interface ExecutionContextBase {
    readonly toolApproval: ApprovalAccessor;
  }
}

// Plugin tool
@Tool({
  name: 'approve_tool',
  description: 'Approve a tool for execution',
  inputSchema: { toolName: z.string() },
})
class ApproveToolTool extends ToolContext {
  async execute(input: { toolName: string }) {
    const accessor = this.get(ApprovalAccessor);
    await accessor.approve(input.toolName);
    return { approved: input.toolName };
  }
}

// Plugin definition
@Plugin({
  name: 'tool-approval',
  description: 'Tool approval system',
  providers: [ApprovalService, ApprovalAccessor],
  exports: [ApprovalAccessor],
  tools: [ApproveToolTool],
  contextExtensions: [
    {
      property: 'toolApproval',
      token: ApprovalAccessor,
      errorMessage: 'ToolApprovalPlugin not installed',
    },
  ],
})
export class ToolApprovalPlugin {}

// Using the plugin
@Tool({
  name: 'dangerous_operation',
  inputSchema: { data: z.string() },
})
class DangerousOperationTool extends ToolContext {
  async execute(input: { data: string }) {
    // Check approval via context extension
    const isApproved = await this.toolApproval.isApproved('dangerous_operation');

    if (!isApproved) {
      return { error: 'This tool requires approval. Please run approve_tool first.' };
    }

    // Proceed with operation
    return { result: 'Operation completed' };
  }
}

@App({
  name: 'secure-app',
  plugins: [ToolApprovalPlugin],
  tools: [DangerousOperationTool],
})
class SecureApp {}

@FrontMcp({
  info: { name: 'Secure Server', version: '1.0.0' },
  apps: [SecureApp],
})
export default class SecureServer {}
```
