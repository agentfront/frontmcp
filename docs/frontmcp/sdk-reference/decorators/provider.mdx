## Basic Usage

```typescript
import { Provider } from '@frontmcp/sdk';

@Provider({ name: 'UserService' })
export class UserService {
  async findById(id: string) {
    return { id, name: 'John Doe' };
  }

  async create(data: { name: string; email: string }) {
    return { id: 'user_123', ...data };
  }
}
```

## Signature

```typescript
function Provider(providedMetadata: ProviderMetadata): ClassDecorator
```

## Configuration Options

| Property      | Type                                                              | Default                | Description                  |
| ------------- | ----------------------------------------------------------------- | ---------------------- | ---------------------------- |
| `name`        | `string` (required)                                               | —                      | Human-readable provider name |
| `id`          | `string`                                                          | —                      | Optional unique identifier   |
| `description` | `string`                                                          | —                      | Optional description         |
| `scope`       | `ProviderScope` (`ProviderScope.GLOBAL \| ProviderScope.CONTEXT`) | `ProviderScope.GLOBAL` | Provider lifetime scope      |

The metadata is strict: any other key (such as `provide` or `useClass`) is rejected when the class is decorated. `ProviderScope.SESSION` and `ProviderScope.REQUEST` are kept as deprecated aliases of `ProviderScope.CONTEXT`.

### Global (Default)

Single instance shared across all requests:

```typescript
import { Provider } from '@frontmcp/sdk';

@Provider({ name: 'ConfigService' }) // scope defaults to ProviderScope.GLOBAL
class ConfigService {
  private readonly config: Record<string, string | undefined> = { ...process.env };

  get(key: string) {
    return this.config[key];
  }
}
```

### Context

One instance per session the server verified, and a new instance on every request for a caller without one (a stateless request, or MCP 2026-07-28):

```typescript
import { Provider, ProviderScope } from '@frontmcp/sdk';

@Provider({ name: 'RequestLogger', scope: ProviderScope.CONTEXT })
class RequestLogger {
  private logs: string[] = [];

  log(message: string) {
    this.logs.push(`[${new Date().toISOString()}] ${message}`);
  }

  getLogs() {
    return this.logs;
  }
}
```

## Using Providers

A decorated class is its own DI token: register it in `providers`, then resolve it with `this.get(TheClass)`.

### In Tools

```typescript
import { Tool, ToolContext, z } from '@frontmcp/sdk';
import { UserService } from './user.service';

@Tool({
  name: 'get_user',
  inputSchema: { userId: z.string() },
})
class GetUserTool extends ToolContext {
  async execute(input: { userId: string }) {
    const userService = this.get(UserService);
    return userService.findById(input.userId);
  }
}
```

### In Resources

```typescript
import { Resource, ResourceContext } from '@frontmcp/sdk';
import { DataService } from './data.service';

@Resource({ name: 'data', uri: 'data://all' })
class DataResource extends ResourceContext {
  async execute(uri: string) {
    const dataService = this.get(DataService);
    return { contents: [{ uri, text: JSON.stringify(dataService.getAll()) }] };
  }
}
```

### In Agents

```typescript
import { Agent, AgentContext, z } from '@frontmcp/sdk';
import { AnalyticsService } from './analytics.service';

@Agent({
  name: 'analyst',
  inputSchema: { question: z.string() },
  llm: { provider: 'openai', model: 'gpt-4o', apiKey: { env: 'OPENAI_API_KEY' } },
})
class AnalystAgent extends AgentContext {
  async execute(input: { question: string }) {
    const analytics = this.get(AnalyticsService);
    analytics.track('question', input.question);
    return super.execute(input); // run the default agent loop
  }
}
```

## Provider Dependencies

Providers can depend on other providers through their constructor. The parameter types are read from the `design:paramtypes` metadata, so the project needs `emitDecoratorMetadata` (and `experimentalDecorators`) enabled:

```typescript
import { Provider } from '@frontmcp/sdk';
import { Pool } from 'pg';

@Provider({ name: 'DatabaseService' })
class DatabaseService {
  private pool = new Pool({ connectionString: process.env.DATABASE_URL });

  query(sql: string, params?: unknown[]) {
    return this.pool.query(sql, params);
  }
}

@Provider({ name: 'UserRepository' })
class UserRepository {
  constructor(private db: DatabaseService) {}

  async findById(id: string) {
    const result = await this.db.query(`SELECT * FROM users WHERE id = $1`, [id]);
    return result.rows[0];
  }
}
```

## Registration

### In Apps

```typescript
@App({
  name: 'my-app',
  providers: [UserService, DatabaseService, UserRepository],
  tools: [GetUserTool],
})
class MyApp {}
```

### In Server

```typescript
@FrontMcp({
  info: { name: 'Server', version: '1.0.0' },
  apps: [MyApp],
  providers: [ConfigService, LoggingService], // Shared across all apps
})
class Server {}
```

### In Plugins

```typescript
@Plugin({
  name: 'cache-plugin',
  providers: [CacheService],
  exports: [CacheService], // also visible to the host app's tools
})
class CachePlugin {}
```

`providers` accepts `@Provider` classes, factory providers (below), and named `{ name, provide, useValue }` and `{ name, provide, useClass }` objects.

## Optional Dependencies

Use `tryGet()` for optional dependencies:

```typescript
@Tool({ name: 'my_tool', inputSchema: {} })
class MyTool extends ToolContext {
  async execute() {
    // Required dependency (throws if not found)
    const required = this.get(RequiredService);

    // Optional dependency (returns undefined if not found)
    const optional = this.tryGet(OptionalService);
    if (optional) {
      optional.doSomething();
    }

    return { ok: true };
  }
}
```

## Factory Providers

A factory provider binds a token to the value its `useFactory` returns. It takes `name` (required, like `@Provider`), `provide` (the token), an optional `inject` function that returns the dependency tokens, `useFactory` (called with the resolved dependencies, in order, and may be `async`), and an optional `scope`:

```typescript
import { App } from '@frontmcp/sdk';

class DatabaseService {
  constructor(readonly url: string) {}
}

@App({
  name: 'app',
  providers: [
    ConfigService,
    {
      name: 'DatabaseService',
      provide: DatabaseService,
      inject: () => [ConfigService] as const,
      useFactory: (config: ConfigService) => new DatabaseService(config.get('DATABASE_URL') ?? ''),
    },
  ],
})
class MyApp {}
```

`inject` is a function, not an array, so it can reference classes declared later in the file. Without `inject`, the factory receives no arguments.

## Provider Tokens

A token can be any class or a `Symbol`. Type a `Symbol` token with `Token<T>` so `this.get()` returns `T`, and bind it with a factory provider. This lets tools depend on an interface instead of an implementation:

```typescript
import { App, Provider, Tool, ToolContext, type Token, z } from '@frontmcp/sdk';

// Interface
export interface CacheStore {
  get(key: string): Promise<string | null>;
  set(key: string, value: string): Promise<void>;
}

// Token
export const CACHE_STORE: Token<CacheStore> = Symbol('CacheStore');

// Implementation
@Provider({ name: 'MemoryCacheStore' })
class MemoryCacheStore implements CacheStore {
  private entries = new Map<string, string>();

  async get(key: string) {
    return this.entries.get(key) ?? null;
  }

  async set(key: string, value: string) {
    this.entries.set(key, value);
  }
}

// Use with token
@Tool({ name: 'cache_lookup', inputSchema: { key: z.string() } })
class CacheLookupTool extends ToolContext {
  async execute(input: { key: string }) {
    const cache = this.get(CACHE_STORE); // typed as CacheStore
    return { value: await cache.get(input.key) };
  }
}

// Bind the token to the implementation
@App({
  name: 'app',
  providers: [
    MemoryCacheStore,
    {
      name: 'CacheStore',
      provide: CACHE_STORE,
      inject: () => [MemoryCacheStore] as const,
      useFactory: (store: MemoryCacheStore) => store,
    },
  ],
  tools: [CacheLookupTool],
})
class CacheApp {}
```

## Static Values

Provide a configuration value with a factory that takes no dependencies (a named `{ name, provide, useValue }` object works too):

```typescript
import { App, type Token } from '@frontmcp/sdk';

export const API_URL: Token<string> = Symbol('API_URL');

@App({
  name: 'app',
  providers: [{ name: 'ApiUrl', provide: API_URL, useFactory: () => 'https://api.example.com' }],
})
class ApiApp {}
```

## Full Example

```typescript
import { Provider, ProviderScope, Tool, ToolContext, App, FrontMcp, type Token } from '@frontmcp/sdk';
import { z } from '@frontmcp/sdk';

// Interface
interface NotificationService {
  send(to: string, message: string): Promise<void>;
}

// Token for the interface
export const NOTIFICATION_SERVICE: Token<NotificationService> = Symbol('NotificationService');

// Provider implementations
@Provider({ name: 'ConfigService' })
class ConfigService {
  private env = process.env;

  get(key: string): string | undefined {
    return this.env[key];
  }

  require(key: string): string {
    const value = this.get(key);
    if (!value) throw new Error(`Missing config: ${key}`);
    return value;
  }
}

@Provider({ name: 'EmailNotificationService' })
class EmailNotificationService implements NotificationService {
  constructor(private config: ConfigService) {}

  async send(to: string, message: string) {
    const apiKey = this.config.require('EMAIL_API_KEY');
    // Send email with apiKey...
    console.log(`Sending email to ${to}: ${message}`);
  }
}

@Provider({ name: 'RequestTracker', scope: ProviderScope.CONTEXT })
class RequestTracker {
  private startTime = Date.now();
  private events: string[] = [];

  track(event: string) {
    this.events.push(`[${Date.now() - this.startTime}ms] ${event}`);
  }

  getTimeline() {
    return this.events;
  }
}

// Tool using providers
@Tool({
  name: 'send_notification',
  inputSchema: {
    recipient: z.string(),
    message: z.string(),
  },
})
class SendNotificationTool extends ToolContext {
  async execute(input: { recipient: string; message: string }) {
    const tracker = this.get(RequestTracker);
    tracker.track('Tool started');

    const notification = this.get(NOTIFICATION_SERVICE);
    await notification.send(input.recipient, input.message);

    tracker.track('Notification sent');

    return {
      success: true,
      timeline: tracker.getTimeline(),
    };
  }
}

@App({
  name: 'notifications',
  providers: [
    ConfigService,
    RequestTracker,
    EmailNotificationService,
    {
      name: 'NotificationService',
      provide: NOTIFICATION_SERVICE,
      inject: () => [EmailNotificationService] as const,
      useFactory: (email: EmailNotificationService) => email,
    },
  ],
  tools: [SendNotificationTool],
})
class NotificationsApp {}

@FrontMcp({
  info: { name: 'Notification Service', version: '1.0.0' },
  apps: [NotificationsApp],
})
export default class NotificationServer {}
```
