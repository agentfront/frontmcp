## Hook Types

| Decorator | Timing                                    | Use Case                                 |
| --------- | ----------------------------------------- | ---------------------------------------- |
| `@Stage`  | One more step of a stage, next to its own | Extra work in the stage, ending the flow |
| `@Will`   | Before stage executes                     | Pre-processing, validation               |
| `@Did`    | After stage executes                      | Post-processing, logging                 |
| `@Around` | Wraps stage execution                     | Caching, error handling                  |

## FlowHooksOf

Create typed hook decorators for a specific flow:

```typescript
import { FlowCtxOf, FlowHooksOf, Provider } from '@frontmcp/sdk';

const { Stage, Will, Did, Around } = FlowHooksOf('tools:call-tool');

@Provider({ name: 'my-hooks' })
class MyHooks {
  @Will('execute')
  async beforeExecute(ctx: FlowCtxOf<'tools:call-tool'>) {
    ctx.logger.info('Before tool execution');
  }

  @Did('execute')
  async afterExecute(ctx: FlowCtxOf<'tools:call-tool'>) {
    ctx.logger.info('After tool execution');
  }
}
```

The stage names a decorator accepts are the stages of that flow (`execute`, `validateInput`, `finalize`, ... for `tools:call-tool`). A hook method receives the running flow, typed `FlowCtxOf<'<flow-name>'>`; see [Hook Context](#hook-context).

## Available Flows

`@frontmcp/sdk` exports the decorators of the common flows ready to use (`const { Will, Did } = ToolHook`):

| Flow Name                           | Description                 | Export                      |
| ----------------------------------- | --------------------------- | --------------------------- |
| `tools:call-tool`                   | Tool execution              | `ToolHook`                  |
| `tools:list-tools`                  | List tools                  | `ListToolsHook`             |
| `resources:read-resource`           | Read resource               | `ResourceHook`              |
| `resources:list-resources`          | List resources              | `ListResourcesHook`         |
| `resources:list-resource-templates` | List resource templates     | `ListResourceTemplatesHook` |
| `prompts:get-prompt`                | Get prompt                  | `PromptHook`                |
| `prompts:list-prompts`              | List prompts                | `ListPromptsHook`           |
| `completion:complete`               | Argument completion         | `CompletionHook`            |
| `agents:call-agent`                 | Agent invocation            | `AgentCallHook`             |
| `jobs:execute-job`                  | One attempt of a job        | `JobHook`                   |
| `channels:send-notification`        | Channel notification        | `ChannelSendHook`           |
| `channels:list`                     | Channels a session receives | `ChannelListHook`           |
| `http:request`                      | HTTP request handling       | `HttpHook`                  |

Up to 1.8.7, `FlowHooksOf('prompts:get-prompt')`, `FlowHooksOf('prompts:list-prompts')` and
`FlowHooksOf('completion:complete')` did not typecheck in a consumer project: those flows' types were missing
from the published type definitions.

## @Will (Before Hook)

Execute before a flow stage:

```typescript
const { Will } = FlowHooksOf('tools:call-tool');

@Provider({ name: 'validation-hooks' })
class ValidationHooks {
  @Will('execute', { priority: 10 })
  async requireArguments(ctx: FlowCtxOf<'tools:call-tool'>) {
    const { arguments: toolArguments } = ctx.state.required.input;
    if (!toolArguments || Object.keys(toolArguments).length === 0) {
      throw new InvalidInputError('Arguments are required');
    }
  }

  @Will('execute', { priority: 20 })
  async logStart(ctx: FlowCtxOf<'tools:call-tool'>) {
    ctx.logger.info(`Starting tool: ${ctx.state.required.input.name}`);
  }
}
```

The tool name is `ctx.state.required.input.name`: the flow state has no `toolName` key, and `ctx.state.get()` accepts only the keys the flow declares.

### Priority

Lower priority values execute first:

```typescript
@Will('execute', { priority: 10 }) // Executes first
@Will('execute', { priority: 20 }) // Executes second
```

### Filter

Conditionally execute hooks:

```typescript
@Will('execute', {
  filter: (ctx) => ctx.state.required.input.name === 'sensitive_tool'
})
async logSensitiveAccess(ctx: FlowCtxOf<'tools:call-tool'>) {
  ctx.logger.warn('Accessing sensitive tool');
}
```

## @Did (After Hook)

Execute after a flow stage. A `Did` hook runs only when the stage succeeded; to see failures too, use an [`@Around`](#around-wrapper-hook) hook.

```typescript
const { Did } = FlowHooksOf('tools:call-tool');

@Provider({ name: 'audit-hooks' })
class AuditHooks {
  @Did('execute')
  async auditToolCall(ctx: FlowCtxOf<'tools:call-tool'>) {
    const { input, toolContext } = ctx.state.required;
    await ctx.get(AuditService).log({
      tool: input.name,
      timestamp: new Date(),
      output: toolContext.output,
    });
  }
}
```

## @Around (Wrapper Hook)

Wrap stage execution for full control. The stage's Will hooks run first, `next()` runs the stage itself, and its Did hooks run after the Around hook returns. An Around hook that does not call `next()` skips the stage. With several Around hooks on one stage, each `next()` runs the next one, and the innermost runs the stage. For `tools:call-tool`, the tool's result is `ctx.state.required.toolContext.output`.

```typescript
const { Around } = FlowHooksOf('tools:call-tool');

@Provider({ name: 'caching-hooks' })
class CachingHooks {
  private readonly cache = new Map<string, unknown>();

  @Around('execute')
  async cacheResults(ctx: FlowCtxOf<'tools:call-tool'>, next: () => Promise<void>) {
    const { name, arguments: toolArguments } = ctx.state.required.input;
    const cacheKey = `${name}:${JSON.stringify(toolArguments)}`;
    const cached = this.cache.get(cacheKey);

    if (cached !== undefined) {
      ctx.state.required.toolContext.output = cached;
      return; // Skip actual execution
    }

    await next(); // Execute the stage

    this.cache.set(cacheKey, ctx.state.required.toolContext.output);
  }
}
```

## @Stage (Stage Step)

Add a step to one of the flow's stages. The stage must be one the flow declares: a flow runs only the stages in its plan, so a `@Stage` hook cannot add a new stage.

A `@Stage` hook does not replace the flow's own step: both run, in `priority` order (lower first). At equal priority a plugin's or provider's step runs before the flow's own, and a step declared on an entry class runs after it. The value the method returns is ignored. To keep the flow's own step from running, end the flow with `ctx.respond(result)` from the `@Stage` hook, or skip the stage with an [`@Around`](#around-wrapper-hook) hook that does not call `next()`.

```typescript
const { Stage } = FlowHooksOf('tools:call-tool');

@Provider({ name: 'maintenance-hooks' })
class MaintenanceHooks {
  @Stage('execute', {
    filter: (ctx) => ctx.state.required.input.name === 'export_report',
  })
  async answerDuringMaintenance(ctx: FlowCtxOf<'tools:call-tool'>) {
    // Runs before the tool's own execute step, which ctx.respond() then skips
    ctx.respond({ content: [{ type: 'text', text: 'Reports are paused for maintenance.' }] });
  }
}
```

`ctx.respond()` ends the flow with that output: the rest of the stage and the stages after it are skipped, while the stage's `Did` hooks and the flow's cleanup stages still run. In `tools:call-tool` the output is a complete `CallToolResult` and does not go through the tool's output schema.

To share a value between the hooks of one call, key it on the flow in a `WeakMap`, or keep it in a local variable of an `@Around` hook: the flow state accepts only the keys the flow declares, so `ctx.state.set('processedInput', ...)` does not compile. The [Full Example](#full-example) times calls that way.

## Hook Options

```typescript
interface HookOptions<Ctx = unknown> {
  priority?: number;                              // Execution order (lower = first)
  filter?: (ctx: Ctx) => boolean | Promise<boolean>; // Conditional execution
  appliesTo?: 'own-app' | 'uncovered-apps';        // Reach of an app plugin's hook (default 'own-app')
}
```

A hook of a plugin installed on an app runs, in `tools/call`, `resources/read`, `prompts/get` and
`completion/complete`, only for that app's entries, unless the plugin has `scope: 'server'`: then its hooks run for
every app's entries. With `appliesTo: 'uncovered-apps'` it also runs for the entries of
any app that has no instance of the same hook (same class and method) of its own or from a server-level plugin. Use it
for gates that an entry's metadata asks for, so the entry is not left ungated when the plugin sits on another app.

## Registering Hooks

### In Apps

```typescript
@App({
  name: 'my-app',
  providers: [AuditHooks, CachingHooks],
  tools: [MyTool],
})
class MyApp {}
```

### In Plugins

```typescript
@Plugin({
  name: 'audit-plugin',
  providers: [AuditHooks],
})
class AuditPlugin {}
```

### On the Server

Hooks declared on a server-level provider run for every app's tools, resources and prompts:

```typescript
@FrontMcp({
  info: { name: 'my-server', version: '1.0.0' },
  apps: [MyApp],
  providers: [AuditHooks],
})
export default class Server {}
```

### On a CONTEXT-Scoped Provider

A hook declared on a provider with `scope: ProviderScope.CONTEXT` runs on the instance built for the request (or
session), the same instance the request's tools resolve with `this.get()`:

```typescript
@Provider({ name: 'request-trace', scope: ProviderScope.CONTEXT })
class RequestTrace {
  readonly steps: string[] = [];

  @Will('execute')
  record(ctx: FlowCtxOf<'tools:call-tool'>) {
    this.steps.push(`tool:${ctx.state.tool?.name}`);
  }
}
```

The provider's class must be known up front (a class provider, not a `useFactory` one). Up to 1.8.7, hooks on
server-level and CONTEXT-scoped providers were silently never registered.

### On an Entry Class

A hook declared on a `@Tool`, `@Resource`, `@Prompt` or `@Agent` class runs only for that entry, on the instance created for each call:

```typescript
@Tool({ name: 'get_ticket', inputSchema: {} })
class GetTicketTool extends ToolContext {
  @Will('execute')
  async recordAccess() {
    this.logger.info('get_ticket called');
  }

  async execute() {
    return { ok: true };
  }
}
```

Because instance methods run on that instance, they can only hook stages from the one that builds it on:
`createToolCallContext`, `createResourceContext`, `createPromptContext` or `createAgentContext` (`Did` and `Stage` on
that stage, and any hook on a later stage).

#### Static hooks for earlier stages

To hook a stage that runs before the instance exists, such as `parseInput`, `findTool`, `checkToolAuthorization` or
`checkEntryAuthorities` (and `findResource`, `findPrompt`, `findAgent`, `checkAgentAuthorization`), declare the hook as
a `static` method. A static hook runs without an instance: `this` is the class, and like every hook it receives the
running flow as its argument, so it reads and changes the flow's `state`, or throws to refuse the call. It joins a call
when the call starts, and only a call for its own entry: a static hook on one tool never runs for another.

```typescript
@Tool({ name: 'transfer', inputSchema: { account: z.string(), amount: z.number() } })
class TransferTool extends ToolContext {
  // Runs before the tool is authorized or its instance is built
  @ToolHook.Will('checkToolAuthorization')
  static refuseFrozen(flowCtx: FlowCtxOf<'tools:call-tool'>) {
    const account = flowCtx.state.input?.arguments?.['account'];
    if (frozenAccounts.has(String(account))) throw new UnauthorizedError('the account is frozen');
  }

  // Rewrites the arguments before the tool is looked up
  @ToolHook.Did('parseInput')
  static normalize(flowCtx: FlowCtxOf<'tools:call-tool'>) {
    const input = flowCtx.state.required.input;
    const account = String(input.arguments?.['account'] ?? '').trim();
    flowCtx.state.set('input', { ...input, arguments: { ...input.arguments, account } });
  }

  async execute({ account, amount }: { account: string; amount: number }) {
    return { account, amount };
  }
}
```

A static hook may also hook the stage that builds the instance (`Will` and `Around` included) or a later stage; it
still runs once per call. Among hooks of the same priority, the class's static hooks run after the hooks of plugins and
providers, as its instance hooks do.

An instance method on an earlier stage, or `Will` or `Around` on the stage that builds the instance, would never run, so
the server refuses to start with `InvalidHookFlowError`, and the message tells you to declare it `static`. A list-flow
hook (`ListToolsHook`, `ListResourcesHook`, `ListPromptsHook`) is refused on an entry class whether it is static or not:
a listing builds no instance and runs for no single entry. Declare list hooks on a provider or a plugin instead. Up to
1.9.3 static hooks on earlier stages were refused too.

A `@Job` class declares hooks for the `jobs:execute-job` flow its attempts run, with the same rules: instance methods
from `createJobContext` on, `static` methods from the first stage. A hook it declares for another flow fails startup
with `InvalidHookFlowError`. See [Jobs: Hooks](/frontmcp/servers/jobs#hooks). Up to 1.9.3 jobs ran outside any flow and
any hook on a `@Job` class failed startup.

Up to 1.8.7, all of these were accepted and silently never ran.

Hooks from an app's providers, and from the plugins (including plugins nested inside them) and plugin providers it lists, run only for that app's tools, resources and prompts (`tools:call-tool`, `resources:read-resource`, `prompts:get-prompt`, `completion:complete`), including the ones its adapters and plugins provide, such as the tools an OpenAPI adapter generates. The hooks of a plugin with `scope: 'server'` are the exception: they run for every app's entries, the hooks declared on its providers included, while the plugin's providers and context extensions stay with the app that installs it. Plugins and providers registered on the server (`@FrontMcp({ plugins, providers })`) apply to every app. Resources and prompts the server serves outside every app, such as the SEP-2640 `skill://` resources, run every app's hooks.

## Hook Context

A hook method receives the running flow, typed `FlowCtxOf<'<flow-name>'>`:

```typescript
import { FlowCtxOf, ToolHook } from '@frontmcp/sdk';

const { Will } = ToolHook;

class InspectHooks {
  @Will('execute')
  async inspect(ctx: FlowCtxOf<'tools:call-tool'>) {
    // Flow state, typed per flow
    const { name, arguments: toolArguments } = ctx.state.required.input;

    // The flow's logger
    ctx.logger.info(`Calling ${name}`, toolArguments);

    // A provider registered on the server
    await ctx.get(AuditService).record(name);
  }
}
```

See [Hook Context](/frontmcp/sdk-reference/registries/hook-registry#hook-context) for every member of the flow.

## Error Handling

A hook that throws stops the flow, and the error reaches the client like an error from the stage itself. An Around hook sees a failing stage as a rejected `next()`. It can rethrow, retry by calling `next()` again, or catch the rejection and return normally. Returning normally handles the failure: the stage counts as successful, so set the output the stage would have produced first. An Around hook skipped by its `filter` still runs the stage:

```typescript
@Will('execute')
async validateInput(ctx: FlowCtxOf<'tools:call-tool'>) {
  // Throwing stops execution and reports error
  throw new InvalidInputError('Validation failed');
}

@Around('execute')
async errorHandler(ctx: FlowCtxOf<'tools:call-tool'>, next: () => Promise<void>) {
  try {
    await next();
  } catch (error) {
    ctx.logger.error('Execution failed', { error });
    // Rethrow to keep the failure; returning instead would handle it
    throw error;
  }
}
```

## Full Example

```typescript
import {
  App,
  FlowCtxOf,
  FlowHooksOf,
  FrontMcp,
  Plugin,
  Provider,
  RateLimitError,
  Tool,
  ToolContext,
  z,
} from '@frontmcp/sdk';

const { Will, Did, Around } = FlowHooksOf('tools:call-tool');

type ToolFlow = FlowCtxOf<'tools:call-tool'>;

// Audit hook provider: the Will and Did hooks of one call share its start time through the flow instance
@Provider({ name: 'audit-hooks' })
class AuditHooks {
  private readonly startTimes = new WeakMap<ToolFlow, number>();
  private auditLog: Array<{ tool: string; timestamp: Date; duration: number }> = [];

  @Will('execute', { priority: 0 })
  async recordStart(ctx: ToolFlow) {
    this.startTimes.set(ctx, Date.now());
  }

  @Did('execute', { priority: 1000 })
  async recordEnd(ctx: ToolFlow) {
    const duration = Date.now() - (this.startTimes.get(ctx) ?? Date.now());
    const toolName = ctx.state.required.input.name;

    this.auditLog.push({
      tool: toolName,
      timestamp: new Date(),
      duration,
    });

    ctx.logger.info(`Tool ${toolName} completed in ${duration}ms`);
  }

  getAuditLog() {
    return this.auditLog;
  }
}

// Rate limiting hook
@Provider({ name: 'rate-limit-hooks' })
class RateLimitHooks {
  private calls = new Map<string, number[]>();
  private maxCallsPerMinute = 60;

  @Will('execute', { priority: 5 })
  async checkRateLimit(ctx: ToolFlow) {
    const toolName = ctx.state.required.input.name;
    const now = Date.now();
    const oneMinuteAgo = now - 60000;

    const recentCalls = (this.calls.get(toolName) || [])
      .filter(t => t > oneMinuteAgo);

    if (recentCalls.length >= this.maxCallsPerMinute) {
      throw new RateLimitError(60);
    }

    recentCalls.push(now);
    this.calls.set(toolName, recentCalls);
  }
}

// Caching hook
@Provider({ name: 'cache-hooks' })
class CacheHooks {
  private cache = new Map<string, { value: unknown; expires: number }>();
  private ttl = 60000; // 1 minute

  @Around('execute', {
    filter: (ctx) => ctx.state.tool?.metadata.annotations?.idempotentHint === true
  })
  async cacheIdempotent(ctx: ToolFlow, next: () => Promise<void>) {
    const { name: toolName, arguments: toolArguments } = ctx.state.required.input;
    const cacheKey = `${toolName}:${JSON.stringify(toolArguments)}`;
    const toolContext = ctx.state.required.toolContext;

    const cached = this.cache.get(cacheKey);
    if (cached && cached.expires > Date.now()) {
      ctx.logger.debug('Cache hit', { toolName });
      toolContext.output = cached.value;
      return;
    }

    await next();

    this.cache.set(cacheKey, {
      value: toolContext.output,
      expires: Date.now() + this.ttl,
    });
  }
}

// Plugin bundling hooks
@Plugin({
  name: 'observability',
  description: 'Audit logging, rate limiting, and caching',
  providers: [AuditHooks, RateLimitHooks, CacheHooks],
})
class ObservabilityPlugin {}

// Sample tool
@Tool({
  name: 'get_data',
  inputSchema: { key: z.string() },
  annotations: { idempotentHint: true },
})
class GetDataTool extends ToolContext {
  async execute(input: { key: string }) {
    return { data: `Value for ${input.key}` };
  }
}

// App using plugin
@App({
  name: 'data-app',
  plugins: [ObservabilityPlugin],
  tools: [GetDataTool],
})
class DataApp {}

@FrontMcp({
  info: { name: 'Observable Server', version: '1.0.0' },
  apps: [DataApp],
})
export default class ObservableServer {}
```
