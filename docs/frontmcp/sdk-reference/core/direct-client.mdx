## Interface

```typescript
interface DirectClient {
  // Tools
  listTools(): Promise<FormattedTools>;
  callTool(name: string, args?: Record<string, unknown>, options?: CallToolOptions): Promise<FormattedToolResult>;

  // Resources
  listResources(): Promise<ListResourcesResult>;
  readResource(uri: string): Promise<ReadResourceResult>;
  listResourceTemplates(): Promise<ListResourceTemplatesResult>;

  // Prompts
  listPrompts(): Promise<ListPromptsResult>;
  getPrompt(name: string, args?: Record<string, string>): Promise<GetPromptResult>;

  // Skills
  searchSkills(query: string, options?: SearchSkillsOptions): Promise<SearchSkillsResult>;
  loadSkills(skillIds: string[], options?: LoadSkillsOptions): Promise<LoadSkillsResult>;
  listSkills(options?: ListSkillsOptions): Promise<ListSkillsResult>;

  // Jobs
  listJobs(options?: ListJobsOptions): Promise<ListJobsResult>;
  executeJob(name: string, input?: Record<string, unknown>, options?: ExecuteJobOptions): Promise<JobExecutionResult>;
  getJobStatus(runId: string): Promise<JobStatusResult>;

  // Workflows
  listWorkflows(options?: ListWorkflowsOptions): Promise<ListWorkflowsResult>;
  executeWorkflow(name: string, input?: Record<string, unknown>, options?: ExecuteWorkflowOptions): Promise<WorkflowExecutionResult>;
  getWorkflowStatus(runId: string): Promise<WorkflowStatusResult>;

  // Info
  getSessionId(): string;
  getClientInfo(): ClientInfo;
  getServerInfo(): Implementation;
  getCapabilities(): ServerCapabilities;
  getDetectedPlatform(): LLMPlatform;

  // Lifecycle
  close(): Promise<void>;
}
```

## Connection Functions

### connect(config, options?)

Create a generic DirectClient.

```typescript
import { connect } from '@frontmcp/sdk';

const client = await connect(config, {
  clientInfo: { name: 'my-agent', version: '1.0.0' },
  authToken: 'jwt-token',
});
```

### connectOpenAI(config, options?)

Create a client with OpenAI tool formatting.

```typescript
import { connectOpenAI } from '@frontmcp/sdk';

const client = await connectOpenAI(config);
const tools = await client.listTools();
// tools are in OpenAI function format
```

### connectClaude(config, options?)

Create a client with Anthropic Claude formatting.

```typescript
import { connectClaude } from '@frontmcp/sdk';

const client = await connectClaude(config);
const tools = await client.listTools();
// tools are in Claude tool format
```

### connectVercelAI(config, options?)

Create a client with Vercel AI SDK formatting.

```typescript
import { connectVercelAI } from '@frontmcp/sdk';

const client = await connectVercelAI(config);
const tools = await client.listTools();
// tools are in Vercel AI format
```

## Connection Options

```typescript
// connect(config, options?: ConnectOptions & { app?: string })
// `app` connects to the endpoint that app has of its own (splitByApp, or a standalone app)
interface ConnectOptions {
  clientInfo?: ClientInfo;
  session?: SessionOptions;
  authToken?: string;
  capabilities?: Partial<ClientCapabilities>;
  onElicitation?: ElicitationHandler;
}

interface SessionOptions {
  id?: string;
  user?: {
    sub?: string;
    name?: string;
    email?: string;
    [key: string]: unknown;
  };
  scopes?: string[]; // the caller's scopes: this.auth.scopes in the tools
}

interface ClientInfo {
  name: string;
  version: string;
}
```

## Tool Operations

### listTools()

List all available tools with platform-specific formatting.

```typescript
const tools = await client.listTools();
// FormattedTools depends on detected platform
```

### callTool(name, args?, options?)

Execute a tool and get platform-formatted result. The OpenAI, Claude, LangChain and Vercel AI formats have no place for a result's `isError`, so with them a failed call rejects with `ToolCallError`: `message` is the tool's error text and `result` the raw `CallToolResult` (pass it to the model as an error, for example Claude's `is_error: true`). The raw format returns the result with `isError`. Up to 1.9.1 the platform formats dropped `isError`.

```typescript
const result = await client.callTool('my_tool', {
  input: 'test',
  count: 5,
});
```

Pass `onProgress` to hear the tool's `this.progress()` updates: the call then carries a progress token, without which `this.progress()` sends nothing and returns `false` (as it always did up to 1.9.2).

```typescript
await client.callTool('import_tickets', { file: 'tickets.csv' }, {
  onProgress: ({ progress, total, message }) => console.log(progress, total, message),
});
```

## Resource Operations

### listResources()

List all static resources. Every page is fetched (a FrontMCP server pages its lists 40 at a time by default), so the result never carries a `nextCursor`.

```typescript
const { resources } = await client.listResources();
resources.forEach(r => console.log(r.uri, r.name));
```

### readResource(uri)

Read a resource by URI.

```typescript
const { contents } = await client.readResource('config://app');
console.log(contents[0].text);
```

### listResourceTemplates()

List all resource templates, following every page like `listResources()`.

```typescript
const { resourceTemplates } = await client.listResourceTemplates();
resourceTemplates.forEach(t => console.log(t.uriTemplate));
```

## Prompt Operations

### listPrompts()

List all prompts, following every page like `listResources()`.

```typescript
const { prompts } = await client.listPrompts();

```

### getPrompt(name, args?)

Get a prompt with arguments.

```typescript
const { messages } = await client.getPrompt('research', {
  topic: 'AI safety',

```

## Job Operations

### listJobs(options?)

List all available jobs.

```typescript
const { jobs, count } = await client.listJobs();
jobs.forEach(j => console.log(j.name, j.description));

// Filter by tags
const filtered = await client.listJobs({ tags: ['notification'] });
```

### executeJob(name, input?, options?)

Execute a job by name. Supports background mode.

```typescript
// Inline execution (waits for completion)
const result = await client.executeJob('analyze-text', {
  text: 'Hello world',
  language: 'en',
});
console.log(result.state); // 'completed'
console.log(result.result); // { sentiment: 'positive', ... }

// Background execution (returns immediately with runId)
const { runId } = await client.executeJob('analyze-text', {
  text: 'Hello world',
}, { background: true });
```

### getJobStatus(runId)

Get the status of a job run.

```typescript
const status = await client.getJobStatus(runId);
console.log(status.state);      // 'running' | 'completed' | 'failed'
console.log(status.result);     // Job result (when completed)
console.log(status.logs);       // Execution logs
console.log(status.attempt);    // Current retry attempt
```

## Workflow Operations

### listWorkflows(options?)

List all available workflows.

```typescript
const { workflows, count } = await client.listWorkflows();
workflows.forEach(w => console.log(w.name, w.trigger, w.steps?.length));
```

### executeWorkflow(name, input?, options?)

Execute a workflow by name.

```typescript
// Inline execution
const result = await client.executeWorkflow('data-pipeline', {
  source: 'api',
});
console.log(result.state);       // 'completed' | 'failed'
console.log(result.stepResults); // Per-step results

// Background execution
const { runId } = await client.executeWorkflow('data-pipeline', {
  source: 'api',
}, { background: true });
```

### getWorkflowStatus(runId)

Get the status of a workflow run with per-step details.

```typescript
const status = await client.getWorkflowStatus(runId);
console.log(status.state);          // 'running' | 'completed' | 'failed'
console.log(status.workflowName);   // Workflow name
console.log(status.stepResults);    // { step1: { state, outputs }, ... }
```

## Elicitation

### onElicitation(handler)

Register or replace the handler for elicitation requests: the questions the tools ask with `this.elicit()` (form or URL mode). Questions reach a client only if it declared the `elicitation` capability when it connected: pass `onElicitation` in the connect options (which declares `form` and `url`), or `capabilities: { elicitation: { form: {} } }`. A client that declared it without a handler declines every question. A client that did not declare it gets the fallback flow, answered with `submitElicitationResult()`. Up to 1.9.1 the questions never reached the handler.

The handler gets the question's `message`, `requestedSchema` and `mode` (and `url` in URL mode), with its `elicitId` and `expiresAt` (a timestamp in milliseconds). Up to 1.9.2 those two were missing.

```typescript
const unsubscribe = client.onElicitation(async (request) => {
  console.log('Server asking:', request.message);

  // Get user input
  const response = await getUserInput(request.requestedSchema);

  return {
    action: 'accept',
    content: response,
  };
});

// Later: unsubscribe()
```

### submitElicitationResult(elicitId, response)

Answer a question from the fallback flow: a tool result whose `_meta.elicitationPending.elicitId` names it. It calls the `sendElicitationResult` tool, which runs the waiting tool again with the answer, and returns that tool's result, formatted as `callTool()` formats it. Up to 1.9.2 it sent an `elicitation/result` request, which failed with `-32601` `Method not found`.

```typescript
const pending = await client.callTool('close_ticket', { id: 'T-1' });
const { elicitId } = pending._meta.elicitationPending;
const result = await client.submitElicitationResult(elicitId, {
  action: 'accept',
  content: { confirmed: true },
});
```

## Lifecycle

### close()

Close the client connection. `connect()` clients of the same config object share one server, which is disposed when the last of them closes; closing one leaves the others' sessions as they are. Up to 1.9.1 closing one disposed the server under the others, clearing their client info. A client from a `DirectMcpServer`'s `connect()` never disposes the server: `server.dispose()` does. Up to 1.9.2 closing one disposed part of it, so `registerTool()` failed and the other clients stopped getting notifications.

```typescript
await client.close();
```

## Usage Examples

### Testing Tools

```typescript
import { connect } from '@frontmcp/sdk';
import config from './server';

describe('Tools', () => {
  let client: DirectClient;

  beforeAll(async () => {
    client = await connect(config);
  });

  afterAll(async () => {
    await client.close();
  });

  test('list tools', async () => {
    const tools = await client.listTools();
    expect(tools.length).toBeGreaterThan(0);
  });

  test('call tool', async () => {
    const result = await client.callTool('get_user', { userId: '123' });
    expect(result.id).toBe('123');
  });
});
```

### OpenAI Integration

```typescript
import { connectOpenAI } from '@frontmcp/sdk';
import OpenAI from 'openai';

const client = await connectOpenAI(config);
const openai = new OpenAI();

// Get tools in OpenAI format
const tools = await client.listTools();

// Use with OpenAI
const completion = await openai.chat.completions.create({
  model: 'gpt-4',
  messages: [{ role: 'user', content: 'List all users' }],
  tools,
});

// Handle tool calls
for (const toolCall of completion.choices[0].message.tool_calls || []) {
  const result = await client.callTool(
    toolCall.function.name,
    JSON.parse(toolCall.function.arguments)
  );
  // Continue conversation with result...
}
```

### Skill Workflow

```typescript
const client = await connect(config);

// Search for relevant skills
const search = await client.searchSkills('data analysis', { limit: 3 });
console.log('Found skills:', search.skills.map(s => s.name));

// Load skill into session
const load = await client.loadSkills([search.skills[0].id], {
  activateSession: true,
  policyMode: 'strict',
});

// Now only skill's tools are available
const tools = await client.listTools();
// tools are filtered to skill's allowlist
```

### With Authentication

```typescript
import { connect } from '@frontmcp/sdk';
import config from './server';

const client = await connect(config, {
  authToken: 'eyJ...',  // JWT token
  session: {
    user: {
      sub: 'user-123',
      name: 'John Doe',
      email: 'john@example.com',
    },
  },
});

// Tools will have access to auth context
const result = await client.callTool('get_my_profile');
```

`session.scopes` gives the tools their caller's scopes (`this.auth.scopes`, `this.auth.hasScope()`), as `authContext.scopes` does for `create()`; without it they're `[]`.

```typescript
const client = await connect(config, {
  session: { user: { sub: 'user-123' }, scopes: ['tickets:write'] },
});
```

## Scope Caching

Connection functions use WeakMap caching to ensure singleton scopes:

```typescript
// Same config object = same scope
const client1 = await connect(config);
const client2 = await connect(config); // Reuses scope

// Different config = different scope
const client3 = await connect({ ...config }); // New scope
```
